## ⚡ 快速开始命令

```bash
# 1. 创建项目
npx create-next-app@latest yisu-hotel --typescript --tailwind --app

# 2. 安装关键依赖
npm install \
  next-auth \
  @next-auth/prisma-adapter \
  prisma \
  @prisma/client \
  react-date-range \
  react-map-gl \
  mapbox-gl \
  geolib \
  stripe \
  @stripe/react-stripe-js

# 3. 初始化 Prisma
npx prisma init

# 4. 启动开发服务器
npm run dev
```
